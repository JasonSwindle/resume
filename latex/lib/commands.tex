%----------------------------------------------------------------------------------------
%	CUSTOM COMMANDS
%----------------------------------------------------------------------------------------

% Roman numerals
\NewDocumentCommand{\RNum}{m}{\uppercase\expandafter{\romannumeral #1\relax}}

% Standard Spacing
\newcommand{\EntryGap}{5pt}
\newcommand{\SectionGap}{10pt}
\newcommand{\RoleGap}{10.5pt}

% Helper for date ranges
% #1: Start Date
% #2: End Date (empty for Present)
\NewDocumentCommand{\formatdaterange}{m m}{
    \ifblank{#2}{\printdate{#1} to Present}{\daterange{#1}{#2}}
}

% Helper for consistent table layout
% #1: Table content
\NewDocumentCommand{\MakeResumeTable}{m}{
    \par\noindent
    \begin{tabularx}{\textwidth}{@{} X @{\hspace{1em}} r @{}}
        #1
    \end{tabularx}\par
}

% Helper for standard entry header
% #1: Primary (Left side, Bold)
% #2: Secondary (Right side)
\NewDocumentCommand{\EntryHeader}{m m}{
    \textbf{#1} & #2 \\
}

% Generic helper for standard resume entries
% #1: Primary (Company/Uni/Award)
% #2: Secondary (Location/Provider)
% #3: Tertiary (Title/Degree/Desc)
% #4: Date Text
\NewDocumentCommand{\RenderEntry}{m m m m}{
    \MakeResumeTable{
        \EntryHeader{#1}{#2}
        #3 & #4
    }
}

% Generic helper for dated items (Roles/Education)
% #1: Name (Company/University)
% #2: Location
% #3: Title/Degree
% #4: Start Date
% #5: End Date
\NewDocumentCommand{\DatedItem}{m m m m m}{
    \RenderEntry{#1}{\formatdaterange{#4}{#5}}{#3}{#2}
}

% Header for a role/job experience
% #1: Company Name
% #2: Location
% #3: Title
% #4: Start Date
% #5: End Date (empty for Present)
\NewDocumentCommand{\roleheader}{m m m m m}{
    \DatedItem{#1}{#2}{#3}{#4}{#5}
}

% Full role/job experience entry: header plus a bullet list.
% #1: Company Name
% #2: Location
% #3: Title
% #4: Start Date
% #5: End Date (empty for Present)
% Body: \item entries (nested itemize allowed)
\NewDocumentEnvironment{role}{m m m m m}{
    \roleheader{#1}{#2}{#3}{#4}{#5}
    \begin{itemize}[beginpenalty=10000]
}{
    \end{itemize}
    \par\addvspace{\RoleGap}
}

% Header for an education entry
% #1: University Name
% #2: Location
% #3: Degree
% #4: Start Date
% #5: End Date (empty for Present)
\NewDocumentCommand{\educationitem}{m m m m m}{
    \DatedItem{#1}{#2}{#3}{#4}{#5}
    \par\addvspace{\EntryGap}
}

% Item for a project
% #1: Project Name
% #2: URL (without scheme, e.g., github.com/user/repo)
% #3: Description
\NewDocumentCommand{\projectitem}{m m m}{
    \MakeResumeTable{
        \EntryHeader{#1}{\href{https://#2}{#2}}
        \multicolumn{2}{@{}p{\textwidth}@{}}{#3}
    }
    \par\addvspace{\EntryGap}
}

% Item for a certification (canonical)
% #1: Certification Name
% #2: Award/Certification Provider
% #3: Description/ID
% #4: Date (YYYY-MM-DD or YYYY)
\ExplSyntaxOn
\NewDocumentCommand{\formatcertdate}{m}{
    \tl_if_in:nnTF {#1} {-} {\printdate{#1}} {#1}
}
\ExplSyntaxOff
\NewDocumentCommand{\certificationitem}{m m m m}{
    \RenderEntry{#1}{#2}{#3}{\formatcertdate{#4}}
    \par\addvspace{\EntryGap}
}

% Backward-compatible alias for certification entries
\NewDocumentCommand{\awarditem}{m m m m}{
    \certificationitem{#1}{#2}{#3}{#4}
}

% Item for a reference
% #1: Quote/Reference Text
% #2: Author Name
\NewDocumentCommand{\refitem}{+m m}{
    \par\noindent
    \begin{minipage}{\textwidth}
        \setlength\parindent{2em}
        \itshape
        ``#1''
    \end{minipage}
    \par\nopagebreak[4]
    \hfill --- \textsc{#2}
    \par\addvspace{\SectionGap}
}

% Skills section: title plus a two-column bullet list.
% #1: Title
% Body: \item entries
\NewDocumentEnvironment{skills}{m}{
    \setlength{\multicolsep}{2pt}
    \setlength{\premulticols}{0pt}
    \setlength{\postmulticols}{0pt}
    \par\noindent\textbf{#1}\par\nopagebreak[4]
    \begin{multicols}{2}
        \begin{itemize}
}{
        \end{itemize}
    \end{multicols}
    \par\addvspace{4pt}
}

% Profile header
% #1: Name
% #2: Email Address
% #3: Country Code
% #4: Phone Number
% #5: Website
% #6: Location
\NewDocumentCommand{\profile}{m m m m m m}{
    {\centering
    \begin{tabular}{c}
        {\Huge \scshape {#1}} \\[\EntryGap]
        \href{mailto:#2}{#2}%
        \ifblank{#4}{}{\quad\phonenumber[country={#3},foreign]{#4}}%
        \ifblank{#6}{}{\quad#6}%
        \ifblank{#5}{}{\quad\url{#5}}
    \end{tabular}\par}%
    \addvspace{4pt}%
}
